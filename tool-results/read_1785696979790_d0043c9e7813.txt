  1037→  const parsed = parseGitHubUrl(url);

  1039→    throw new Error(
  1040→      `Invalid GitHub URL: "${url}". Expected format: https://github.com/owner/repo`,
  1041→    );
  1042→  }
  1043→
  1044→  const { owner, repo } = parsed;
  1045→  const aiProvider = createAIProvider({ maxTokens: 8192, temperature: 0.3 });
  1046→  const startTime = Date.now();
  1047→
  1048→  // Try AI-powered analysis first
  1049→  let result = await aiPoweredAnalysis(url, owner, repo, aiProvider);
  1050→
  1051→  // Fall back to heuristic-based analysis if AI fails or returns incomplete data
  1052→  if (!result) {
  1053→    const elapsed = Date.now() - startTime;
  1054→    result = generateHeuristicAnalysis(url, owner, repo, elapsed);
  1055→  } else {
  1056→    // Ensure durationMs reflects total wall-clock time
  1057→    result = { ...result, durationMs: Date.now() - startTime };
  1058→  }
  1059→
  1060→  return result;
  1061→}
