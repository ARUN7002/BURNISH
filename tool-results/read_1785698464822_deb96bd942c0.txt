    31→/** Simple deterministic hash from a string → 0–1 */
    32→function hashSeed(input: string): number {
    33→  let h = 0;
    34→  for (let i = 0; i < input.length; i++) {
    35→    h = ((h << 5) - h + input.charCodeAt(i)) | 0;
    36→  }
    37→  return (Math.abs(h) % 10000) / 10000;
    38→}
    39→
    40→/** Derive a deterministic float 0–max from a seed string and index */
    41→function seededScore(seed: string, index: number, min: number, max: number): number {
    42→  const v = hashSeed(`${seed}:${index}`);
    43→  return Math.round((min + v * (max - min)) * 10) / 10;
    44→}
    45→
    46→/** Pick an item from an array deterministically based on seed */
    47→function seededPick<T>(seed: string, index: number, arr: T[]): T {
    48→  const i = Math.floor(hashSeed(`${seed}:pick:${index}`) * arr.length);
    49→  return arr[i];
    50→}

    52→function scoreToGrade(score: number): 'A+' | 'A' | 'B' | 'C' | 'D' | 'F' {
    53→  if (score >= 95) return 'A+';
    54→  if (score >= 85) return 'A';
    55→  if (score >= 70) return 'B';
    56→  if (score >= 55) return 'C';
    57→  if (score >= 40) return 'D';
    58→  return 'F';
    59→}
    60→
    61→function scoreToStatus(score: number): 'excellent' | 'good' | 'fair' | 'poor' | 'critical' {
    62→  if (score >= 90) return 'excellent';
    63→  if (score >= 75) return 'good';
    64→  if (score >= 60) return 'fair';
    65→  if (score >= 40) return 'poor';
    66→  return 'critical';
    67→}
    68→
    69→function scoreToMaturityLevel(score: number): 'Initial' | 'Managed' | 'Defined' | 'Quantitatively Managed' | 'Optimizing' {
    70→  if (score >= 90) return 'Optimizing';
    71→  if (score >= 75) return 'Quantitatively Managed';
    72→  if (score >= 55) return 'Defined';
    73→  if (score >= 35) return 'Managed';
    74→  return 'Initial';
    75→}

    78→  return Math.max(min, Math.min(max, v));

    81→function parseGitHubUrl(url: string): { owner: string; repo: string } | null {
    82→  const patterns = [
    83→    /^https?:\/\/github\.com\/([^/]+)\/([^/]+?)(?:\.git)?\s*$/,
    84→    /^https?:\/\/www\.github\.com\/([^/]+)\/([^/]+?)(?:\.git)?\s*$/,
    85→    /^github\.com\/([^/]+)\/([^/]+?)(?:\.git)?\s*$/,
    86→    /^([^/]+)\/([^/]+)$/,
    87→  ];
    88→  for (const p of patterns) {
    89→    const m = url.trim().match(p);
    90→    if (m) return { owner: m[1], repo: m[2].replace(/\.git$/, '') };
    91→  }
    92→  return null;
    93→}

   107→  const dependencyManagement = clamp(seededScore(seed, 6, 40, 88));
   108→
   109→  const overall = clamp(
   110→    (codeQuality * 0.2 + testing * 0.18 + documentation * 0.12 +
   111→      security * 0.2 + maintainability * 0.15 + dependencyManagement * 0.15),
   112→  );
   113→
   114→  return {
   115→    overall,
   116→    breakdown: { codeQuality, testing, documentation, security, maintainability, dependencyManagement },
   117→    grade: scoreToGrade(overall),

   178→  const overall = clamp(dimensions.reduce((s, d) => s + (d.level / d.maxLevel) * 100, 0) / dimensions.length);
   179→
   180→  return {
   181→    overall,
   182→    dimensions,
   183→    level: scoreToMaturityLevel(overall),

   475→  const allRisks: Omit<SecurityRisk, 'severity'>[] = [

   477→      category: 'Dependency Vulnerabilities',
   478→      finding: 'Outdated dependencies with known CVEs detected in production dependency tree',
   479→      recommendation: 'Run a full dependency audit, upgrade affected packages, and implement automated CVE scanning in CI',
   480→    },
   481→    {
   482→      category: 'Secrets Exposure',
   483→      finding: 'API keys or credentials potentially exposed in repository history or configuration files',
   484→      recommendation: 'Rotate all potentially compromised credentials, use a secrets manager, and enable push protection',
   485→    },

   492→      category: 'Authentication',
   493→      finding: 'Authentication mechanisms may not implement rate limiting or brute-force protection',
   494→      recommendation: 'Implement rate limiting, account lockout policies, and consider MFA enforcement',
   495→    },
   496→    {
   497→      category: 'Data Protection',
   498→      finding: 'Sensitive data may not be encrypted at rest or in transit for all data paths',
   499→      recommendation: 'Audit data flows, enforce TLS everywhere, and enable encryption at rest for PII',
   500→    },
   501→    {
   502→      category: 'Access Control',
   503→      finding: 'Role-based access control may not be enforced consistently across all endpoints',
   504→      recommendation: 'Implement centralized RBAC middleware and audit all endpoints for proper authorization checks',
   505→    },
   506→  ];
   507→
   508→  const severities: Array<'critical' | 'high' | 'medium' | 'low'> = ['critical', 'high', 'medium', 'low'];
   509→  const count = 4 + Math.floor(hashSeed(`${seed}:secCount`) * 3);
   510→  const used = new Set<number>();
   511→  const items: SecurityRisk[] = [];
   512→
   513→  for (let i = 0; i < count && i < allRisks.length; i++) {
   514→    let idx: number;
   515→    do {
   516→      idx = Math.floor(hashSeed(`${seed}:sec:${i}`) * allRisks.length);
   517→    } while (used.has(idx) && used.size < allRisks.length);
   518→    used.add(idx);
   519→
   520→    const sIdx = Math.floor(hashSeed(`${seed}:secSev:${idx}`) * 10);
   521→    const severity = sIdx < 2 ? 'critical' : sIdx < 4 ? 'high' : sIdx < 7 ? 'medium' : 'low';
   522→
   523→    items.push({ ...allRisks[idx], severity });
   524→  }
   525→
   526→  return items;
   527→}

  1030→export async function analyzeRepository(url: string): Promise<AnalysisResult> {
  1031→  const parsed = parseGitHubUrl(url);

  1033→    throw new Error(

  1035→    );
  1036→  }
  1037→
  1038→  const { owner, repo } = parsed;
  1039→  const aiProvider = createAIProvider({ maxTokens: 8192, temperature: 0.3 });
  1040→  const startTime = Date.now();
  1041→
  1042→  // Try AI-powered analysis first
  1043→  let result = await aiPoweredAnalysis(url, owner, repo, aiProvider);
  1044→
  1045→  // Fall back to heuristic-based analysis if AI fails or returns incomplete data
  1046→  if (!result) {
  1047→    const elapsed = Date.now() - startTime;
  1048→    result = generateHeuristicAnalysis(url, owner, repo, elapsed);
  1049→  } else {
  1050→    // Ensure durationMs reflects total wall-clock time
  1051→    result = { ...result, durationMs: Date.now() - startTime };
  1052→  }
  1053→
  1054→  return result;
